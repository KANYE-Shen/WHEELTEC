cmake_minimum_required(VERSION 3.17)
project(wheeltec_gui_sync LANGUAGES CXX)
cmake_policy(SET CMP0100 NEW)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(ignition-gui6 REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Quick)
add_library(WheeltecRenderSync SHARED RenderSync.cc RenderSync.hh wheeltec_sync.qrc)
target_link_libraries(WheeltecRenderSync PRIVATE ignition-gui6::ignition-gui6 Qt5::Quick)
