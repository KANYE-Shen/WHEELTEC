cmake_minimum_required(VERSION 3.16)
project(wheeltec_slam)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(slam_toolbox REQUIRED)
find_package(Boost REQUIRED COMPONENTS serialization thread system filesystem)
find_library(ASYNC_SLAM_LIBRARY async_slam_toolbox HINTS "${slam_toolbox_DIR}/../../../lib" REQUIRED)
find_library(COMMON_SLAM_LIBRARY toolbox_common HINTS "${slam_toolbox_DIR}/../../../lib" REQUIRED)
find_library(KARTO_SLAM_LIBRARY kartoSlamToolbox HINTS "${slam_toolbox_DIR}/../../../lib" REQUIRED)
add_executable(motion_gated_slam src/motion_gated_slam.cpp)
target_include_directories(motion_gated_slam PRIVATE include)
ament_target_dependencies(motion_gated_slam slam_toolbox)
target_link_libraries(motion_gated_slam ${ASYNC_SLAM_LIBRARY} ${COMMON_SLAM_LIBRARY} ${KARTO_SLAM_LIBRARY} ${Boost_LIBRARIES})
install(TARGETS motion_gated_slam DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_motion_gate test/test_motion_gate.cpp)
  target_include_directories(test_motion_gate PRIVATE include)
  add_test(NAME motion_gate COMMAND test_motion_gate)
endif()
ament_package()
